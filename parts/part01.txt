const STORAGE_KEY = "semiannual-bonus-data-v1";

// ====== 雲端同步設定（Firebase Realtime Database）======
// 把你 Firebase 專案的 Database URL 貼在這裡（結尾是 firebaseio.com/ 或 .firebasedatabase.app/），
// 貼上之後大家打開「同一個網址／同一份檔案」就能共用同一份資料。
// 如果先不填（維持空字串），系統會照舊只存在這台電腦的瀏覽器裡。
const FIREBASE_URL = "";
const PALETTE = ["#A63D2E","#A9862F","#3F6D4E","#3C5A78","#7A5C3E","#5B5573","#8B5E3C","#4F6B52"];
const DEFAULT_ROLES = ["工程負責人","外機系統負責人","內機系統負責人","排水負責人","線型負責人","風管負責人"];

function periodFromDate(d){
  const m = d.getMonth()+1; const y = d.getFullYear();
  if(m>=4 && m<=9) return `AS-${y}`;
  if(m>=10) return `OS-${y}`;
  return `OS-${y-1}`;
}
function periodLabel(id){
  const [type,yStr] = id.split("-"); const y = Number(yStr);
  if(type==="AS") return `${y} 年 4月～9月（${y+1} 年 1月結算）`;
  return `${y} 年 10月～${y+1} 年 3月（${y+1} 年 7月結算）`;
}
function periodShort(id){
  const [type,yStr] = id.split("-"); const y = Number(yStr);
  return type==="AS" ? `${y}年上半期(4-9月)` : `${y}年下半期(10-3月)`;
}
function nextPeriodId(id){
  const [type,yStr] = id.split("-"); const y = Number(yStr);
  return type==="AS" ? `OS-${y}` : `AS-${y+1}`;
}
function fyYearOf(id){ const [,yStr] = id.split("-"); return Number(yStr); }
function fyLabel(y){ return `${y} 年度（${y} 年4月 ～ ${y+1} 年3月）`; }

// Empty starter state for a brand-new period (AS-2026, no slips yet).
// This is only used the FIRST time the tool runs on a browser with no saved data yet —
// once anything is saved, that saved data always takes priority and this is never touched.
const IMPORTED_SEED_DATA = {
  "order": ["AS-2026"],
  "current": "AS-2026",
  "roster": [],
  "approvers": [],
  "periods": {
    "AS-2026": { "slips": [] }
  }
};

function freshImportedData(){
  return JSON.parse(JSON.stringify(IMPORTED_SEED_DATA));
}

function ensureShape(obj){
  const out = obj && typeof obj === "object" ? obj : freshImportedData();
  if(!Array.isArray(out.order)) out.order = [];
  if(!out.periods || typeof out.periods !== "object") out.periods = {};
  if(!Array.isArray(out.roster)) out.roster = [];
  if(!Array.isArray(out.approvers)) out.approvers = [];
  if(!out.current || !out.periods[out.current]) out.current = out.order[0] || periodFromDate(new Date());
  if(!out.periods[out.current]){
    out.order.push(out.current);
    out.periods[out.current] = { slips: [] };
  }
  Object.keys(out.periods).forEach(id=>{
    if(!Array.isArray(out.periods[id].slips)) out.periods[id].slips = [];
    out.periods[id].slips.forEach((s,i)=>{
      if(!s.id) s.id = `${id}-${Date.now()}-${i}`;
      if(!Array.isArray(s.models)) s.models = [];
      if(!Array.isArray(s.roles)) s.roles = DEFAULT_ROLES.map(role=>({role,name:""}));
      if(s.approver == null) s.approver = "";
      if(s.note == null) s.note = "";
    });
  });
  return out;
}

function loadLocalData(){
  try{
    const raw = localStorage.getItem(STORAGE_KEY);
    if(!raw) return freshImportedData();
    return ensureShape(JSON.parse(raw));
  }catch(err){
    console.warn("local load failed", err);
    return freshImportedData();
  }
}

function saveLocalData(){
  try{ localStorage.setItem(STORAGE_KEY, JSON.stringify(data)); }
  catch(err){ console.warn("local save failed", err); }
}

async function fetchCloudState(){
  if(!FIREBASE_URL) return { ok:false, reason:"no-url" };
  const url = FIREBASE_URL.replace(/\/$/, "") + "/state.json";
  try{
    const res = await fetch(url, { method:"GET", cache:"no-store" });
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    const payload = await res.json();
    return { ok:true, data: payload ? ensureShape(payload) : null };
  }catch(err){
    console.warn("cloud load failed", err);
    return { ok:false, reason:String(err) };
  }
}

async function pushCloudState(){
  if(!FIREBASE_URL) return { ok:false, reason:"no-url" };
  const url = FIREBASE_URL.replace(/\/$/, "") + "/state.json";
  try{
    const res = await fetch(url, {
      method:"PUT",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify(data)
    });
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    return { ok:true };
  }catch(err){
    console.warn("cloud save failed", err);
    return { ok:false, reason:String(err) };
  }
}

async function initApp(){
  render();
  if(FIREBASE_URL){
    const cloud = await fetchCloudState();
    if(cloud.ok){
      cloudMode = true;
      if(cloud.data){
        data = cloud.data;
        saveLocalData();
      }else{
        await pushCloudState();
      }
      render();
    }else{
      renderHeader();
    }
  }
}

function persist(){
  saveLocalData();
  renderHeader();
  if(!cloudMode) return;
  clearTimeout(cloudSaveTimer);
  cloudSaveTimer = setTimeout(async ()=>{
    cloudSaving = true;
    renderHeader();
    const result = await pushCloudState();
    cloudSaving = false;
    if(!result.ok) cloudMode = false;
    renderHeader();
  }, 350);
}

function fmt(n){ return Number(n||0).toLocaleString("zh-TW"); }
function esc(s){ return String(s??"").replace(/[&<>"']/g, c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c])); }
function slipBonus(s){ return Math.round((Number(s.revenue)||0) * (Number(s.sharePercent)||0) / 100); }
function currentSlips(){ return data.periods[data.current]?.slips || []; }
function currentPeriod(){ return data.periods[data.current]; }

function summarize(slips){
  const totalRevenue = slips.reduce((sum,s)=>sum+(Number(s.revenue)||0),0);
  const totalBonus = slips.reduce((sum,s)=>sum+slipBonus(s),0);
  return { totalRevenue, totalBonus, count:slips.length };
}

function personSummary(slips){
  const map = new Map();
  slips.forEach(s=>{
    const name = s.person || "未指定";
    if(!map.has(name)) map.set(name,{revenue:0,bonus:0,count:0});
    const x = map.get(name);
    x.revenue += Number(s.revenue)||0;
    x.bonus += slipBonus(s);
    x.count += 1;
  });
  return map;
}

function renderBackupMenu(){
  const menu = document.getElementById("backupMenu");
  if(!menu) return;
  menu.className = backupMenuOpen ? "open" : "";
  const slots = [1,2,3].map(slot=>{
    let meta = null;
    try{ meta = JSON.parse(localStorage.getItem(`${STORAGE_KEY}-backup-${slot}-meta`)||"null"); }catch(e){}
    const label = meta ? `${meta.label}｜${new Date(meta.time).toLocaleString("zh-TW")}` : "空白";
    return `<div class="bitem ${meta?'clickable':''}" data-load-backup="${slot}">備份 ${slot}：${esc(label)}</div>`;
  }).join("");
  menu.innerHTML = `<div class="bsave" id="manualSaveBtn">＋ 儲存到下一個備份槽</div>${slots}`;
  const saveBtn = document.getElementById("manualSaveBtn");
  if(saveBtn) saveBtn.onclick = ()=>{
    let slot = 1;
    const metas = [1,2,3].map(i=>{
      try{return JSON.parse(localStorage.getItem(`${STORAGE_KEY}-backup-${i}-meta`)||"null");}catch(e){return null;}
    });
    const empty = metas.findIndex(x=>!x);
    if(empty>=0) slot=empty+1;
    else slot = metas.map((x,i)=>({i,time:x?.time||0})).sort((a,b)=>a.time-b.time)[0].i+1;
    localStorage.setItem(`${STORAGE_KEY}-backup-${slot}`, JSON.stringify(data));
    localStorage.setItem(`${STORAGE_KEY}-backup-${slot}-meta`, JSON.stringify({label:periodShort(data.current),time:Date.now()}));
    backupMenuOpen=false; renderHeader();
  };
  menu.querySelectorAll("[data-load-backup]").forEach(el=>{
    el.onclick = ()=>{
      const slot = Number(el.getAttribute("data-load-backup"));
      const raw = localStorage.getItem(`${STORAGE_KEY}-backup-${slot}`);
      if(!raw) return;
      if(!confirm(`確定載入備份 ${slot}？目前尚未另外備份的變更會被覆蓋。`)) return;
      data = ensureShape(JSON.parse(raw));
      backupMenuOpen=false; persist(); render();
    };
  });
}
